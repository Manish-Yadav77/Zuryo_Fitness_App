import axios from "axios";
import { env } from "@/config/env";
import { tokenStorage } from "@/lib/storage";
import type { FieldErrors } from "@/lib/validation";

export const AUTH_LOGOUT_EVENT = "zuryo:auth-logout";

export const api = axios.create({
  baseURL: env.apiUrl,
  timeout: 30_000, // generous: a free Render server can take a while to wake up
});

// Attach the JWT to every request.
api.interceptors.request.use((config) => {
  const token = tokenStorage.get();
  if (token) config.headers.Authorization = `Bearer ${token}`;
  return config;
});

// If the server says our token is no longer valid, tell the app to log out.
api.interceptors.response.use(
  (response) => response,
  (error: unknown) => {
    if (
      axios.isAxiosError(error) &&
      error.response?.status === 401 &&
      tokenStorage.get()
    ) {
      window.dispatchEvent(new Event(AUTH_LOGOUT_EVENT));
    }
    return Promise.reject(error);
  },
);

interface ApiErrorBody {
  error?: {
    code?: string;
    message?: string;
    details?: Array<{ path: string; message: string }>;
  };
}

export interface ParsedApiError {
  message: string;
  code: string | null;
  fieldErrors: FieldErrors;
}

/** Converts anything thrown by an API call into a friendly message + per-field errors. */
export function parseApiError(err: unknown): ParsedApiError {
  if (axios.isAxiosError<ApiErrorBody>(err)) {
    if (err.code === "ECONNABORTED" || err.code === "ETIMEDOUT") {
      return {
        message:
          "The server is taking too long to respond. Please try again in a moment.",
        code: "TIMEOUT",
        fieldErrors: {},
      };
    }
    if (!err.response) {
      return {
        message:
          "Cannot reach the server. Check your connection and try again.",
        code: "NETWORK_ERROR",
        fieldErrors: {},
      };
    }
    const body = err.response.data?.error;
    const fieldErrors: FieldErrors = {};
    for (const d of body?.details ?? []) {
      if (d.path && !(d.path in fieldErrors)) fieldErrors[d.path] = d.message;
    }
    return {
      message: body?.message ?? "Something went wrong. Please try again.",
      code: body?.code ?? null,
      fieldErrors,
    };
  }
  return {
    message: "Something went wrong. Please try again.",
    code: null,
    fieldErrors: {},
  };
}

import { Bot, CheckCircle, Dumbbell, Sparkles, Heart } from 'lucide-react';
import { recordApi } from '@/api/record.api';
import type { AiWorkoutRecommendation } from '@/types/record';
import { parseApiError } from '@/api/client';
import { Alert } from '@/components/ui/Alert';
import { Badge } from '@/components/ui/Badge';
import { Button } from '@/components/ui/Button';
import { Card } from '@/components/ui/Card';

export function AiRecommendationCard() {
  const [recommendation, setRecommendation] = useState<AiWorkoutRecommendation | null>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  async function handleGenerate() {
    if (loading) return;
    setError(null);
    setLoading(true);
    try {
      let data: AiWorkoutRecommendation;
      try {
        data = await recordApi.getAiRecommendation();
      } catch (first) {
        // One silent retry for network blips / server restart (tsx watch, Render wake-up).
        const parsed = parseApiError(first);
        if (parsed.code === 'NETWORK_ERROR' || parsed.code === 'TIMEOUT') {
          await new Promise((r) => setTimeout(r, 2000));
          data = await recordApi.getAiRecommendation();
        } else {
          throw first;
        }
      }
      setRecommendation(data);
    } catch (err) {
      setError(parseApiError(err).message);
    } finally {
      setLoading(false);
    }
  }

  return (
    <Card className="space-y-4 border-emerald-200 bg-linear-to-br from-emerald-50/50 to-white">
      <div className="flex items-center justify-between">
        <div className="flex items-center gap-2">

            <Sparkles className="h-4 w-4" />

            <h2 className="text-base font-bold text-slate-900">AI Workout Coach</h2>
            <p className="text-xs text-slate-500">Personalized routine tailored to your goals & history.</p>
          </div>
        </div>
        <Badge tone="emerald">
          {recommendation ? `${recommendation.source} (${recommendation.latencyMs}ms)` : 'Multi-provider AI'}

      </div>

      {error && <Alert tone="error">{error}</Alert>}
      {loading && <p className="text-center text-xs text-slate-500">Generating your planâ€¦ this can take up to 30 seconds.</p>}

      {!recommendation ? (
        <div className="py-4 text-center">
          <p className="text-xs text-slate-600 mb-4">
            Generates a tailored 60-minute workout from your fitness profile and past sessions.
          </p>
          <Button onClick={handleGenerate} loading={loading}>
            <Bot className="h-4 w-4" /> Generate AI Recommendation
          </Button>
        </div>
      ) : (
        <div className="space-y-4 pt-2">
          <div className="rounded-xl bg-white p-4 shadow-xs border border-emerald-100">
            <div className="flex items-center justify-between">
              <h3 className="font-bold text-slate-900 text-sm">{recommendation.workoutTitle}</h3>
              <Badge tone="sky">{recommendation.focusArea}</Badge>
            </div>

===== FILE: Frontend\src\components\auth\ProtectedRoute.tsx =====
import { Navigate, Outlet } from 'react-router-dom'
import { useAuth } from '@/hooks/useAuth'
import { FullPageSpinner } from '@/components/ui/Spinner'
import { roleHome } from '@/config/nav'
import type { Role } from '@/types/auth'

/**
 * Blocks pages for logged-out users. Pass `roles` to also restrict by role.
 * Example: <ProtectedRoute roles={['admin']} />
 */
export function ProtectedRoute({ roles }: { roles?: Role[] }) {
  const { user, isLoading } = useAuth()

  if (isLoading) return <FullPageSpinner label="Checking your session..." />
  if (!user) return <Navigate to="/login" replace />
  if (roles && !roles.includes(user.role)) return <Navigate to={roleHome[user.role]} replace />

  return <Outlet />
}

import type { AppNotification } from '@/types/notification';

export function NotificationBell() {
  const [notifications, setNotifications] = useState<AppNotification[]>([]);
  const [open, setOpen] = useState(false);
  const dropdownRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    let isMounted = true;
    const fetchNotifications = () => {
      notificationsApi
        .getMine()
        .then((data) => {
          if (isMounted) setNotifications(data || []);
        })
        .catch(console.error);
    };

    fetchNotifications();
    const interval = setInterval(fetchNotifications, 10000);

    return () => {
      isMounted = false;
      clearInterval(interval);
    };
  }, []);

  // Close dropdown when clicking outside
  useEffect(() => {
    function handleClickOutside(e: MouseEvent) {
      if (dropdownRef.current && !dropdownRef.current.contains(e.target as Node)) {
        setOpen(false);
      }
    }

    if (open) {
      document.addEventListener('mousedown', handleClickOutside);
    }
    return () => {
      document.removeEventListener('mousedown', handleClickOutside);
    };
  }, [open]);

  const unreadCount = notifications.filter((n) => !n.readAt).length;

  async function handleMarkRead(id: string) {
    try {
      await notificationsApi.markRead(id);
      setNotifications((prev) =>
        prev.map((n) =>
          n._id === id ? { ...n, readAt: new Date().toISOString() } : n
        )
      );
    } catch (err) {
      console.error(err);
    }
  }

  async function handleMarkAllRead() {
    const unread = notifications.filter((n) => !n.readAt);
    if (unread.length === 0) return;

    try {
      await Promise.all(unread.map((n) => notificationsApi.markRead(n._id)));
      const now = new Date().toISOString();
      setNotifications((prev) =>
        prev.map((n) => ({ ...n, readAt: n.readAt || now }))
      );
    } catch (err) {
      console.error(err);
    }
  }

  return (
    <div className="relative" ref={dropdownRef}>
      <button
        type="button"
        onClick={() => setOpen((s) => !s)}
        className="relative rounded-xl p-2 text-slate-600 transition hover:bg-slate-100 hover:text-slate-900"

              )}

import { useEffect, useState } from 'react';
import { AlertTriangle, CheckCircle2, Clock, Play, Navigation } from 'lucide-react';
import { sessionApi } from '@/api/session.api';
import type { SessionTicker } from '@/types/session';

import { Alert } from '@/components/ui/Alert';
import { Badge } from '@/components/ui/Badge';
import { Button } from '@/components/ui/Button';
import { Card } from '@/components/ui/Card';

interface SessionTimerCardProps {
  bookingId: string;
  isTrainer: boolean;
  bookingStatus: string;
  onStatusChange?: () => void;
}

export function SessionTimerCard({ bookingId, isTrainer, bookingStatus, onStatusChange }: SessionTimerCardProps) {
  const [ticker, setTicker] = useState<SessionTicker | null>(null);
  const [remainingSeconds, setRemainingSeconds] = useState<number>(0);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  async function fetchTicker() {
    try {
      const data = await sessionApi.getTicker(bookingId);
      setTicker(data);
      if (data?.remainingSeconds !== undefined) {
        setRemainingSeconds(data.remainingSeconds);
      }
    } catch (err) {
      console.error(err);
    }
  }

  // Poll server every 5 seconds to maintain authoritative sync
  useEffect(() => {
    fetchTicker();
    const serverSyncInterval = setInterval(fetchTicker, 5_000);
    return () => clearInterval(serverSyncInterval);
  }, [bookingId]);

  // Tick local timer every 1 second for a smooth display
  useEffect(() => {
    if (!ticker || ticker.isCompleted || remainingSeconds <= 0) return;

    const timer = setInterval(() => {
      setRemainingSeconds((prev) => Math.max(0, prev - 1));
    }, 1000);

    return () => clearInterval(timer);
  }, [ticker?.isCompleted, remainingSeconds]);

  async function handleStart() {
    setError(null);
    setLoading(true);
    try {
      await sessionApi.start(bookingId);
      await fetchTicker();
      if (onStatusChange) onStatusChange();
    } catch (err) {
      setError(parseApiError(err).message);
    } finally {
      setLoading(false);
    }
  }

  async function handleOnTheWay() {
    setError(null);
    setLoading(true);
    try {
      await sessionApi.onTheWay(bookingId);
      onStatusChange?.();
    } catch (err) {
      setError(parseApiError(err).message);
    } finally {
      setLoading(false);
    }
  }

  async function handleComplete() {
    setError(null);
    setLoading(true);
    try {
      await sessionApi.complete(bookingId);
      await fetchTicker();
      if (onStatusChange) onStatusChange();
    } catch (err) {
      setError(parseApiError(err).message);
    } finally {
      setLoading(false);
    }
  }

  function formatTime(seconds: number) {
    const mins = Math.floor(seconds / 60);
    const secs = seconds % 60;
    return `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
  }

  if ((bookingStatus === 'CONFIRMED' || bookingStatus === 'ON_THE_WAY') && isTrainer) {
    return (
      <Card className="space-y-4 border-emerald-200 bg-emerald-50/30">
        <div className="flex items-center justify-between">
          <div className="flex items-center gap-2 font-semibold text-slate-900">
            <Clock className="h-5 w-5 text-emerald-600" />
            <span>Workout Timer</span>
          </div>
          <Badge tone="emerald">Ready to Start</Badge>
        </div>
        {error && <Alert tone="error">{error}</Alert>}
        <p className="text-xs text-slate-600">Start session when you arrive at the residence.</p>
        {bookingStatus === 'CONFIRMED' && (
          <Button onClick={handleOnTheWay} loading={loading} variant="secondary" fullWidth>
            <Navigation className="h-4 w-4" /> Mark on the way

import { useCallback, useEffect, useMemo, useState } from 'react'
import type { ReactNode } from 'react'
import axios from 'axios'
import { authApi } from '@/api/auth.api'
import { AUTH_LOGOUT_EVENT } from '@/api/client'
import { tokenStorage } from '@/lib/storage'
import { AuthContext } from './auth-context'
import type { AuthContextValue } from './auth-context'
import type { LoginInput, SignupInput, User } from '@/types/auth'

export function AuthProvider({ children }: { children: ReactNode }) {
  const [user, setUser] = useState<User | null>(null)
  // If a token is saved we must verify it before deciding "logged in or not".
  const [isLoading, setIsLoading] = useState<boolean>(() => tokenStorage.get() !== null)

  // 1) On first load: restore the session from the saved token.
  useEffect(() => {
    if (tokenStorage.get() === null) return
    let cancelled = false

    authApi
      .me()
      .then((u) => {
        if (!cancelled) setUser(u)
      })
      .catch((err: unknown) => {
        // Only throw the token away if the server says it is invalid.
        // A network error (server asleep) must not log the user out.
        if (axios.isAxiosError(err) && err.response?.status === 401) tokenStorage.clear()
      })
      .finally(() => {
        if (!cancelled) setIsLoading(false)
      })

    return () => {
      cancelled = true
    }
  }, [])

  // 2) If any API call gets a 401 later, the axios client fires this event.
  useEffect(() => {
    const onForcedLogout = () => {
      tokenStorage.clear()
      setUser(null)
    }
    window.addEventListener(AUTH_LOGOUT_EVENT, onForcedLogout)
    return () => window.removeEventListener(AUTH_LOGOUT_EVENT, onForcedLogout)
  }, [])

  const login = useCallback(async (input: LoginInput) => {
    const { token, user: loggedIn } = await authApi.login(input)
    tokenStorage.set(token)
    setUser(loggedIn)
    return loggedIn
  }, [])

  const signup = useCallback(async (input: SignupInput) => {
    const { token, user: created } = await authApi.register(input)
    tokenStorage.set(token)
    setUser(created)
    return created
  }, [])

  const logout = useCallback(() => {
    tokenStorage.clear()
    setUser(null)
  }, [])

  const value = useMemo<AuthContextValue>(
    () => ({ user, isLoading, isAuthenticated: user !== null, login, signup, logout }),
    [user, isLoading, login, signup, logout],
  )

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>
}

function minutesToTimeStr(mins: number): string {
  const h = Math.floor(mins / 60) % 24;
  const m = mins % 60;
  return `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}`;
}

function timeStrToMinutes(timeStr: string): number {
  if (!timeStr) return 0;
  const [h, m] = timeStr.split(':').map(Number);
  return (h || 0) * 60 + (m || 0);
}

function formatMinutesTo12H(mins: number): string {
  const h24 = Math.floor(mins / 60) % 24;
  const m = mins % 60;
  const period = h24 >= 12 ? 'PM' : 'AM';
  const h12 = h24 % 12 === 0 ? 12 : h24 % 12;
  return `${h12}:${String(m).padStart(2, '0')} ${period}`;
}

  const { user } = useAuth();
  const { id } = useParams();
  const [booking, setBooking] = useState<Booking | null>(null);
  const [record, setRecord] = useState<SessionRecord | null>(null);
  const [loading, setLoading] = useState(true);
  const [cancelling, setCancelling] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [success, setSuccess] = useState<string | null>(null);
  const [paying, setPaying] = useState(false);
  const [rating, setRating] = useState(5);
  const [feedback, setFeedback] = useState('');
  const [savingFeedback, setSavingFeedback] = useState(false);
  const [feedbackSubmitted, setFeedbackSubmitted] = useState(false);

  async function loadBooking() {
    if (!id) return;
    try {
      const [bookingData, recordData] = await Promise.all([
        bookingsApi.getById(id),
        recordApi.getByBooking(id).catch(() => null),
      ]);
      setBooking(bookingData);
      setRecord(recordData);

      if (recordData?.customerRating) {
        setRating(recordData.customerRating);
        setFeedback(recordData.customerFeedback || '');
        setFeedbackSubmitted(true);
      }
    } catch (err) {
      setError(parseApiError(err).message);
    } finally {
      setLoading(false);
    }
  }

  useEffect(() => {
    loadBooking();
  }, [id]);

  async function handleCancel() {
    if (!id) return;
    setError(null);
    setCancelling(true);
    try {
      const updated = await bookingsApi.cancel(id, 'Cancelled via web dashboard');
      setBooking(updated);
      setSuccess('Booking cancelled successfully.');
    } catch (err) {
      setError(parseApiError(err).message);
    } finally {
      setCancelling(false);
    }
  }

  async function handlePay() {
    if (!booking) return;
    setError(null);
    setPaying(true);
    try {
      const order = await paymentsApi.createOrder(booking._id);
      if (!(await loadRazorpayScript()) || !window.Razorpay) {
        throw new Error('Could not load the payment window.');
      }
      new window.Razorpay({
        key: order.keyId || env.razorpayKeyId,
        amount: order.amountPaise,
        currency: 'INR',
        name: 'Zuryo Fit',
        description: 'Trainer session',
        order_id: order.orderId,
        prefill: { name: user?.name, email: user?.email },
        theme: { color: '#059669' },
        handler: async (response: RazorpaySuccessResponse) => {
          try {
            await paymentsApi.verifyPayment({
              orderId: response.razorpay_order_id,
              paymentId: response.razorpay_payment_id,
              signature: response.razorpay_signature,
            });
            setSuccess('Payment confirmed. Your trainer has been notified.');
            loadBooking();
          } catch (err) {
            setError(parseApiError(err).message);
          } finally {
            setPaying(false);
          }
        },
        modal: { ondismiss: () => setPaying(false) },
      }).open();
    } catch (err) {
      setError(parseApiError(err).message);
      setPaying(false);
    }
  }

    setError(null);

      const updatedRecord = await recordApi.submitFeedback(booking._id, rating, feedback);

      setError(parseApiError(err).message);

      setSavingFeedback(false);

        return 'amber';

        return 'rose';
      default:
        return 'slate';
    }
  };

function format12H(timeStr: string): string {
  if (!timeStr) return '';
  const [h, m] = timeStr.split(':').map(Number);
  if (isNaN(h) || isNaN(m)) return timeStr;
  const period = h >= 12 ? 'PM' : 'AM';
  const h12 = h % 12 === 0 ? 12 : h % 12;
  return `${h12}:${String(m).padStart(2, '0')} ${period}`;
}

### 6.3 How trainer assignment works (location and tie-breaking)

Code: `assignment.service.ts` (`findBestTrainerForSlot`, `rankCandidates`).

1. **Candidate pool:** all active trainers whose `zoneIds` contain the society's `zoneId`.
2. **Eligibility** for the requested 60-minute slot, evaluated per trainer, with a recorded reason:
   - the slot lies inside one of the trainer's weekly working windows (evaluated in India time, so server timezone does not matter);
   - the slot start is not inside a time-off block;
   - none of the four 15-minute atoms of the slot is already locked for that trainer.
3. **Location:** distance in km between the society's coordinates and the trainer's base location using the **Haversine formula** (`utils/geo.ts`). Locations are seeded (simulated GPS).
4. **Preferred trainer:** if the customer picked one and that trainer is eligible, they win (`preferredHonored=true`). If not eligible, assignment silently falls back to ranking.
5. **Ranking (strict order, not a weighted score):**
   1. smaller distance,
   2. fewer sessions already booked that day (load balancing),
   3. higher rating,
   4. trainer id string order (a deterministic final tie-break so results are repeatable).
6. **Audit:** every attempt writes an `AssignmentDecision` containing all candidates, their distance, load, rating, eligibility and the reason for rejection, plus the chosen trainer or `NO_TRAINER`. Admin sees this in the console.
7. **No trainer:** the API returns `409 NO_TRAINER_AVAILABLE` (and logs a `NO_TRAINER` decision). The UI shows an error and the slot grid shows an empty state when nothing is available.

### 6.4 Preventing double booking and handling simultaneous requests

- Every session is split into four **15-minute atoms**. Booking inserts four `SlotLock` rows `(trainerId, atomStart)` inside one **MongoDB transaction** together with the `Booking` and `AssignmentDecision`.
- A **unique index** on `(trainerId, atomStart)` means that if two requests choose the same trainer for overlapping time, the database rejects the second insert (`E11000`). The transaction aborts and the API returns `409 SLOT_ALREADY_BOOKED`. Overlaps that are offset by 15 minutes still share at least one atom, so they collide too.
- **Payment hold:** card bookings are created as `PENDING_PAYMENT` with locks in state `HELD` and `holdExpiresAt = now + 10 min`. A background job marks unpaid bookings `EXPIRED` and deletes held locks. Payment verification only confirms a booking whose hold is still valid.
- **Idempotency:** the client sends an `Idempotency-Key` header. The server returns the existing booking for the same customer and key, and a unique sparse index backs this up.
- **Customer overlap:** a customer cannot hold two active bookings in overlapping time.
- **Honest note:** the front end generates a fresh key on every submit click, so protection against a user double-clicking comes from the customer-overlap check and the slot locks, not from the key alone. The key protects automatic retries that reuse it.

### 6.5 Session timers, reminders and status reliability

- **Authoritative times:** when the trainer starts a session, the server atomically moves the booking `CONFIRMED/ON_THE_WAY â†’ IN_PROGRESS` and stores `startedAt`, `warnAt = +55 min`, `endsAt = +60 min` using the **server clock**. The conditional update means a double-click cannot start it twice.
- **Ticker endpoint:** `GET /bookings/:id/session/ticker` returns `serverTime`, `startedAt`, `endsAt`, `remainingSeconds` and flags. The browser sets its countdown from `remainingSeconds`, ticks locally each second for smoothness, and **re-syncs from the server every 5 seconds**.
- **Browser refresh:** nothing important lives in the browser. The page reloads and fetches the ticker again.
- **App closed:** state is in the database. A background ticker (`ticker.service.ts`, runs at boot and every 30 s) fires the 55-minute warning and the completion event exactly once, using conditional updates (`warnFiredAt` / `completeFiredAt`), so multiple server instances cannot double-fire.
- **Device clock differences:** the client never uses its own clock to decide state; it only counts seconds between server syncs. A wrong device clock cannot shorten or extend a session.
- **Delayed notifications:** warning and completion can be late by up to about 30 seconds (the ticker interval). The status is still correct because the ticker endpoint also **completes lazily on read** when `endsAt` has passed.
- **Known limitation:** on a free Render instance the process sleeps when idle, so background events pause until the next request. The warning is intentionally skipped if the server wakes after `endsAt`. Mitigations: always-on instance (about $7/month) or an external pinger on `/api/health`; production fix is a durable job queue (section 13).

### 6.6 Pricing, commission and package balances, securely

- **Rules live in the database** (`PricingConfig`): normal rate, peak rate, commission percent, peak windows (weekday list + start/end minute, **India time**), and a `version` that increments on every admin edit.
- **How peak is identified:** a slot is peak when its **start time** (converted to Asia/Kolkata) falls inside any peak window; start is inclusive and end exclusive. Defaults: weekdays 06:00â€“09:00 and 17:00â€“21:00, weekends 07:00â€“11:00. (`peak.ts`, `pricing.service.ts`)
- **Calculation:** `buildQuote` is a pure function working in integer paise. Commission = `round(session Ã— percentÃ—100 / 10000)`. Example normal: 39900 + 3990 = **43890**. Example peak: 49900 + 4990 = **54890**. 12.5% commission rounds to the nearest paisa (tested).
- **Snapshot:** the quote (tier, amounts, percent, config version) is stored on the booking, so later rule changes never alter an existing booking.
- **No client tampering:** the request body contains no amount. `POST /payments/order` reads `booking.priceSnapshot.totalPaise` and creates the Razorpay order from that. Package orders use the template's price from the database.
- **Payment verification:** `HMAC-SHA256(secret, orderId|paymentId)` compared with a constant-time function. Only one concurrent verify can move the payment `CREATED â†’ PAID` (atomic conditional update). Repeating the same verify returns the same result and changes nothing.
- **Package balance:** a package is created only after a verified payment, and a unique index on `paymentId` guarantees **one package per payment**. Using a credit runs a conditional atomic update (`usedSessions < totalSessions`, not expired, soonest-expiring first) and writes a `PackageLedger` `CONSUME` row inside the booking transaction. The unique `(bookingId, type)` index means a booking can never deduct twice.
- **Cancellation and refunds (policy):** cancelling at least **2 hours** before start, or any cancellation made by the trainer, is eligible. For package bookings, a `RESTORE` ledger row is written and `usedSessions` decreases by one (also idempotent). For paid bookings the payment is marked `REFUNDED`. Cancelling inside 2 hours gives no refund or credit. Unpaid (`PENDING_PAYMENT`) bookings simply release their slot.
- **Simulated refund:** in this MVP the refund is recorded in the database only (`demo_refund_â€¦` id). **Production:** call the Razorpay refund API with a stored refund id, confirm through webhook, and surface refund failures in `refundError`.
- **Commission on packages:** commission is added to single-session checkout as the assignment describes. Package bundles are sold at their listed bundle price with no extra commission (an assumption, see section 12).

### 6.7 Notifications, GPS and real-time ETA in production

**Implemented:** persisted in-app notifications (assigned, cancelled, rescheduled, on the way, session started, 55-minute warning, completed), shown in a bell with unread count.

**Production design (proposed, section 14.1 and 14.2):** push through FCM/APNs and WhatsApp/SMS for reminders, delivered by a queue-backed worker; trainer app sends location pings over WebSocket or HTTPS to a location service; ETA from a routing API with traffic, recomputed on a schedule and on significant movement; customer receives updates over a WebSocket channel.

### 6.8 Authentication, authorization, secrets and privacy

- **Passwords:** bcrypt, 12 rounds; unknown-email logins compare against a dummy hash to keep response time similar (reduces user-enumeration by timing). Password rules: 8â€“72 characters with a letter and a number.
- **Tokens:** JWT signed with `JWT_SECRET` (at least 32 characters, enforced), 12-hour expiry, carrying only `sub` and `role`.
- **Roles:** public signup always creates a `customer`; trainers and admins exist only through the seed or admin tooling. Routes are guarded with `requireAuth` + `requireRole`.
- **Ownership (object-level authorization):** `assertCanViewBooking` lets only the owning customer, the assigned trainer or an admin read a booking, ticker or record. Others receive **404, not 403**, so booking ids cannot be probed.
- **Brute force:** login and register are rate-limited (30 attempts per 15 minutes per IP).
- **Transport and headers:** Helmet defaults; CORS allow-list from `CLIENT_ORIGIN`; JSON body limit 100 kB; `trust proxy` set for Render so rate limits see real client IPs.
- **Validation:** Zod on every body, ObjectId format checks on ids (malformed id returns a clean 404, tested).
- **Secrets:** only in environment variables (Render / local `.env`), validated at boot; the Razorpay secret and AI keys never reach the browser. `.env` is git-ignored.
- **Privacy:** synthetic data only. Personal data collected: name, email, phone, flat number, age, weight and a free-text limitations field. Production would add explicit consent, data minimisation, retention limits, encryption at rest (Atlas default) and a deletion workflow in line with India's DPDP Act 2023.
- **Known gaps:** token is kept in `localStorage` (XSS exposure; an httpOnly cookie is safer); `isActive` is re-checked only on `/auth/me`, so a deactivated user's token works until expiry on other routes; no refresh-token rotation; no global rate limit beyond auth.

### 6.9 Scaling plan

| Pressure | Change | Why |
| :-- | :-- | :-- |
| More API traffic | Run several stateless API instances behind the host's load balancer | JWT is stateless; the ticker uses conditional updates so duplicates are safe |
| Availability queries are the heaviest read (a loop per slot per trainer) | Precompute availability per trainer/day, cache in Redis, batch the lock query into one `$in` query | Removes N+1 database calls |
| Background work (ticker, notifications) | Move to a queue (BullMQ + Redis or SQS) with delayed jobs scheduled at `warnAt` and `endsAt`, plus a separate worker service | Exact-time events, retries, no dependence on the API process staying awake |
| Real-time updates | Replace polling with WebSockets / Server-Sent Events (Socket.IO with Redis adapter) | Fewer requests, instant UI updates |
| Database growth | Atlas dedicated tier, read replicas for admin/reporting, archive old bookings, add indexes for new queries | Isolate analytics from booking writes |
| Many societies/cities | Zone-based sharding of assignment queries, geospatial index (`2dsphere`) on trainer location instead of loop + Haversine | Faster candidate search |
| Payments | Razorpay webhooks + reconciliation job | Handles browser-closed-after-payment cases |
| Observability | Structured logs, error tracking, uptime checks, metrics | Operate it reliably |

## 7. The AI feature: personalised workout recommendation

**Endpoint:** `GET /api/records/ai-recommendation` (customer only). **Code:** `Backend/src/services/ai.service.ts`. **UI:** `AiRecommendationCard.tsx` on the customer dashboard.

**How it works**
1. Loads the customer's latest assessment and their last 5 session records.
2. Builds one prompt: role (certified fitness coach), the profile (age, weight, goals, level, preferences, limitations), the recent sessions (duration, exercises with sets and reps, rating, trainer notes, suggested next workout), and a required JSON shape: `workoutTitle`, `focusArea`, `warmUp[]`, `mainCircuit[{exercise,sets,reps}]`, `coolDown[]`, `coachNote`.
3. Tries **NVIDIA NIM** (OpenAI-compatible chat completions, `temperature 0.3`, JSON response mode), then **Google Gemini** (`generateContent` with JSON mime type), each with an **8-second timeout**; any error or bad JSON moves to the next option.
4. Last fallback is a **deterministic rules engine** (level-based circuit, goal-based title) so the user always gets a plan.
5. The response includes `source` (`NVIDIA` / `GEMINI` / `RULES`), `model` and `latencyMs`; the UI shows them.

| Models | NVIDIA model from `NVIDIA_MODEL` (set in the hosting dashboard; `.env.example` suggests `meta/llama-3.2-90b-vision-instruct`), Gemini `gemini-2.5-flash` from `LLM_MODEL`, rules baseline |
| Credentials | Server-side environment variables only |
| Approx. cost per interaction | Rules: $0. ## 12. Assumptions and known limitations

**Assumptions**
- Currency INR; all business rules evaluated in India Standard Time.
- Society commission is added to single-session checkout; packages are sold at bundle price without extra commission.
- Trainer locations are fixed seeded base points; there is no live GPS.
- Cancellation policy: free cancellation/credit restore at least 2 hours before start; trainer-initiated cancellation always refunds.

**Known limitations**
1. Refunds are simulated in the database (no Razorpay refund API call).
2. No Razorpay webhook. If a customer pays but closes the browser before `/payments/verify`, the booking can expire while the payment stays captured. Production needs webhook handling plus an auto-refund/reconciliation job. Likewise, a verified payment arriving after the hold expired marks the payment `FAILED` without refunding.
3. Reschedule has no UI button yet; the API exists and is rule-guarded. It also keeps the original price snapshot (no re-pricing) and deletes then re-inserts locks outside a transaction.
4. The trainer may start a session at any time once it is `CONFIRMED` (no check against the scheduled start window) and there is no no-show handling.
5. Notifications are in-app only and polled; the 55-minute "buzzer" is visual, not audible.
6. Background ticker runs inside the API process: it pauses when a free instance sleeps; events can be late by up to 30 s.
7. Expired payment holds keep blocking a slot until the ticker releases them (up to about 30 s after expiry).
8. The trainer workout form saves one exercise per submit (a later save replaces earlier exercises); the data model supports many.
9. Admin "gross revenue" sums every booking's total regardless of status (including unpaid and cancelled), so it is an indicator, not accounting.
10. Admin booking table shows trainer ids rather than names.
11. `Society.commissionPercent` exists in the schema but pricing uses the single global commission from `PricingConfig`.
12. Several availability queries loop per slot per trainer (acceptable for a handful of trainers, not for scale).
13. Some day-boundary calculations (for "sessions today" and the slot grid) use the server's local time zone while eligibility uses India time. This is correct when the server runs in India time and needs the fix described in the submission notes when it runs in UTC.
14. Token in `localStorage`, `isActive` checked only on `/auth/me`, no refresh tokens (section 6.8).
15. Tests: unit and HTTP-security only; the concurrency test is a placeholder (section 11).
16. `routes/ticker.routes.ts` is an empty mounted router (dead code); the real ticker endpoint is `GET /bookings/:id/session/ticker`.

## 14. Design proposals (assignment Â§7)

Detailed designs for **three** features (live location and ETA, surge pricing, push/reminders), then shorter designs for the rest and for market-specific ideas.

### 14.1 Live trainer location and arrival ETA *(detailed)*

**UX:** After "On the way", the customer sees a map pin and "Arriving in 7 min"; the trainer app shows a "Share my location" indicator. Location sharing starts at "On the way" and stops at "Start session".

**Architecture:** trainer mobile app sends a ping every 10â€“15 s (HTTPS or WebSocket) â†’ location service validates and stores the latest point in Redis (`GEOADD`/hash with TTL) â†’ ETA worker calls a routing API (Google Distance Matrix / Mapbox / Ola Maps) at most every 30â€“60 s or when the trainer moves more than a threshold â†’ pushes `eta:update` to the customer over a WebSocket channel (fallback: polling `GET /bookings/:id/tracking`).

**APIs**
- `POST /trainer/location` `{ bookingId, lat, lng, accuracy, heading, ts }` â†’ `204`
- `GET /bookings/:id/tracking` â†’ `{ trainerLocation, etaSeconds, distanceMeters, updatedAt, status }`
- WebSocket event `tracking:update` (room per booking, authorised by booking ownership)

**Data model changes:** `Booking.tracking { active, lastLocation{lat,lng,ts}, etaSeconds, etaUpdatedAt }`; a short-lived `LocationPing` collection with a TTL index (24â€“72 h) for dispute review; consent flag on `TrainerProfile`.

**Dependencies:** native background-location permission (iOS/Android), routing API key, Redis, WebSocket server, battery-friendly update policy.

**Edge cases:** GPS drift or stale pings (drop points older than 60 s, show "last seen"); trainer offline (fall back to scheduled ETA and notify support after N minutes); mock-location spoofing (check accuracy, speed plausibility, optional geofence check at "Start session"); privacy (share only during the active window, never after the session, delete pings on schedule); poor routing results inside gated societies (use gate coordinates as destination); API cost control (throttle by time and distance); customer reschedules mid-trip.

### 14.2 Sophisticated peak and surge pricing *(detailed)*

**Goal:** price from demand and lead time, not just fixed windows, while staying predictable and capped.

**Inputs:** time-of-day windows (already implemented), **lead time** (instant bookings cost more), **demand ratio** = open bookings Ã· available trainer capacity for that zone and hour, day type (weekend/festival/rain), trainer scarcity.

**Formula (example):** `price = base(tier) Ã— clamp(1 + aÂ·(demandRatioâˆ’0.7)âº + bÂ·instantFlag, 1.0, 1.5)`, rounded to â‚¹5 or â‚¹10, then commission as today. Caps and floors come from config; the multiplier is **frozen into the booking's price snapshot at quote time** and honoured for the 10-minute payment hold.

**APIs**
- `GET /pricing/quote?slot=â€¦&societyId=â€¦` â†’ returns tier, multiplier, reasons (`["PEAK_WINDOW","HIGH_DEMAND"]`), and `quoteId` + `expiresAt`
- `PUT /pricing/surge-config` (admin): cap, thresholds, per-zone overrides, on/off
- `POST /bookings` accepts `quoteId`; server rejects expired or mismatched quotes

**Data model changes:** `PricingConfig.surge { enabled, maxMultiplier, demandThresholds[], leadTimeRules[], zoneOverrides[] }`; `Quote` collection (id, slot, society, multiplier, reasons, expiresAt) or signed quote token; `DemandSnapshot` (zone, hour, bookings, capacity) written by a scheduled job; `Booking.priceSnapshot.multiplier, reasons[]`.

**Dependencies:** accurate capacity per zone/hour (from availability), a scheduled aggregation job, admin analytics, clear UI copy ("Higher demand right now").

**Edge cases:** price changes between quote and pay (quote locking with expiry); fairness and trust (show the reason, hard cap, no surge for existing packages' credits); gaming (users toggling slots to find cheaper prices â€” rate limit quotes); regulatory/consumer-protection expectations around transparent pricing; interaction with discounts, promos and package credits (credits never surge); cancellations after surge (refund the snapshot price); data cold-start (fall back to window-based pricing).

### 14.3 Push notifications, reminders and offers *(detailed)*

**UX:** reminders 24 h and 30 min before a session; "trainer assigned / on the way / arrived"; 55-minute warning and completion even when the app is closed; rebooking nudges after a completed session; offers only with consent and quiet hours.

**Architecture:** event â†’ `NotificationService.enqueue()` â†’ queue (BullMQ/Redis) with **delayed jobs** (scheduled at `startAt âˆ’ 30 min`, `warnAt`, `endsAt`) â†’ worker picks channels by preference and fallback order (push â†’ WhatsApp â†’ SMS) â†’ provider adapters (FCM/APNs, WhatsApp Business API, SMS) â†’ delivery receipts update the `Notification` record. Jobs are cancelled or re-scheduled when a booking is cancelled or rescheduled.

**APIs**
- `POST /devices` `{ token, platform }` (register/refresh push token); `DELETE /devices/:id`

**Data model changes:** `DeviceToken(userId, token, platform, lastSeenAt)`; `NotificationPreference(userId, channels, quietHours, marketingOptIn)`; `Notification` gains `channel`, `status` (`QUEUED/SENT/DELIVERED/FAILED`), `dedupeKey`, `scheduledFor`; `Campaign` for offers.
