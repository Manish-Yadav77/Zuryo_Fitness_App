import { z } from 'zod';

const envSchema = z
  .object({
    NODE_ENV: z.enum(['development', 'test', 'production']).default('development'),
    PORT: z.coerce.number().int().positive().default(5000),
    MONGODB_URI: z.string().min(1, 'MONGODB_URI is required'),
    JWT_SECRET: z.string().min(32, 'JWT_SECRET must be at least 32 characters'),
    JWT_EXPIRES_IN_SECONDS: z.coerce.number().int().positive().default(60 * 60 * 12),
    CLIENT_ORIGIN: z.string().min(1).default('http://localhost:5173'),
    SEED_PASSWORD: z.string().min(8).default('Demo@12345'),

    RAZORPAY_KEY_ID: z.string().optional(),
    RAZORPAY_KEY_SECRET: z.string().optional(),

    LLM_API_KEY: z.string().optional(),
    LLM_MODEL: z.string().default('gemini-3.8-flash'),
    NVIDIA_API_KEY: z.string().optional(),
    NVIDIA_MODEL: z.string().default('z-ai/glm-5.3-flash'),
    NVIDIA_BASE_URL: z.string().default('https://integrate.api.nvidia.com/v1'),
  })
  .superRefine((value, ctx) => {
    if (value.NODE_ENV === 'production' && (!value.RAZORPAY_KEY_ID || !value.RAZORPAY_KEY_SECRET)) {
      ctx.addIssue({
        code: 'custom',
        path: ['RAZORPAY_KEY_ID'],
        message: 'RAZORPAY_KEY_ID and RAZORPAY_KEY_SECRET are required in production',
      });
    }
  });

const cleaned = Object.fromEntries(Object.entries(process.env).filter(([, value]) => value !== ''));

const parsed = envSchema.safeParse(cleaned);

if (!parsed.success) {
  console.error('Invalid environment variables:');
  for (const issue of parsed.error.issues) {
    console.error(`  - ${issue.path.join('.')}: ${issue.message}`);
  }
  process.exit(1);
}

export const env = {
  ...parsed.data,
  clientOrigins: parsed.data.CLIENT_ORIGIN.split(',').map((o) => o.trim()).filter(Boolean),
  isProduction: parsed.data.NODE_ENV === 'production',
};

import type { RequestHandler } from 'express';
import bcrypt from 'bcryptjs';
import { z } from 'zod';
import { User, toPublicUser } from '../models/User.js';
import { AppError } from '../utils/AppError.js';
import { signToken } from '../utils/jwt.js';

const BCRYPT_ROUNDS = 12;
// Used so a login for an unknown email takes about as long as a real one (prevents user enumeration by timing).
const DUMMY_HASH = bcrypt.hashSync('not-a-real-password', BCRYPT_ROUNDS);

const emailSchema = z.string().trim().toLowerCase().email('Enter a valid email address');

const registerSchema = z.object({
  name: z.string().trim().min(2, 'Name must be at least 2 characters').max(60, 'Name is too long'),
  email: emailSchema,
  password: z
    .string()
    .min(8, 'Password must be at least 8 characters')
    .max(72, 'Password is too long')
    .regex(/[A-Za-z]/, 'Password must include a letter')
    .regex(/[0-9]/, 'Password must include a number'),
});

const loginSchema = z.object({
  email: emailSchema,
  password: z.string().min(1, 'Password is required').max(72),
});

/** Public signup. The role is ALWAYS customer. Trainers and admins are created by seed/admin only. */
export const register: RequestHandler = async (req, res) => {
  const { name, email, password } = registerSchema.parse(req.body);

  const exists = await User.exists({ email });
  if (exists) {
    throw new AppError(409, 'EMAIL_TAKEN', 'An account with this email already exists', [
      { path: 'email', message: 'This email is already registered' },
    ]);
  }

  const passwordHash = await bcrypt.hash(password, BCRYPT_ROUNDS);
  const user = await User.create({ name, email, passwordHash, role: 'customer' });

  res.status(201).json({
    token: signToken({ sub: user._id.toString(), role: user.role }),
    user: toPublicUser(user),
  });
};

export const login: RequestHandler = async (req, res) => {
  const { email, password } = loginSchema.parse(req.body);

  const user = await User.findOne({ email }).select('+passwordHash');
  const ok = await bcrypt.compare(password, user?.passwordHash ?? DUMMY_HASH);

  if (!user || !ok || !user.isActive) {
    throw new AppError(401, 'INVALID_CREDENTIALS', 'Invalid email or password');
  }

  res.json({
    token: signToken({ sub: user._id.toString(), role: user.role }),
    user: toPublicUser(user),
  });
};

export const me: RequestHandler = async (req, res) => {
  const user = await User.findById(req.auth?.id);
  if (!user || !user.isActive) {
    throw new AppError(401, 'INVALID_TOKEN', 'Your session is no longer valid');
  }
  res.json({ user: toPublicUser(user) });
};

export const getAssignedBookings: RequestHandler = async (req, res) => {
  if (!req.auth) throw new AppError(401, 'UNAUTHENTICATED', 'Log in required');

  const bookings = await Booking.find({ trainerId: req.auth.id }).populate('societyId').sort({ startAt: -1 }).lean();

  const customerIds = [...new Set(bookings.map((b) => b.customerId))];
  const bookingIds = bookings.map((b) => b._id);

  const [profiles, users, records] = await Promise.all([
    CustomerProfile.find({ userId: { $in: customerIds } }).lean(),
    User.find({ _id: { $in: customerIds } }).select('name email').lean(),
    SessionRecord.find({ bookingId: { $in: bookingIds } }).lean(),
  ]);

  const profileMap = new Map(profiles.map((p) => [p.userId, p]));
  const userMap = new Map(users.map((u) => [u._id.toString(), u]));
  const recordMap = new Map(records.map((r) => [r.bookingId.toString(), r]));

  const enrichedBookings = bookings.map((b) => {
    const profile = profileMap.get(b.customerId);
    const user = userMap.get(b.customerId);
    const record = recordMap.get(b._id.toString());

    return {
      ...b,
      customerDetails: {
        name: user?.name,
        flatNumber: profile?.flatNumber,
        phone: profile?.phone,
      },
      sessionRecord: record
        ? {
            durationMinutes: record.durationMinutes,
            exercises: record.exercises,
            trainerNotes: record.trainerNotes,
            suggestedNextWorkout: record.suggestedNextWorkout,
            customerRating: record.customerRating,
            customerFeedback: record.customerFeedback,
          }
        : null,
    };
  });

  res.json({ bookings: enrichedBookings });
};

import type { RequestHandler } from 'express';
import { z } from 'zod';
import { SessionRecord } from '../models/SessionRecord.js';
import { Booking } from '../models/Booking.js';
import { generateWorkoutRecommendation } from '../services/ai.service.js';
import { AppError } from '../utils/AppError.js';
import { requireObjectId } from '../utils/ids.js';
import { assertCanViewBooking } from '../utils/ownership.js';

const createRecordSchema = z.object({
  bookingId: z.string().min(1),
  durationMinutes: z.number().min(1).max(240).default(60),
  exercises: z.array(
    z.object({
      name: z.string().min(1),
      sets: z.number().min(1),
      reps: z.number().optional(),
      weightKg: z.number().optional(),
    }),
  ),
  trainerNotes: z.string().max(2000).optional(),
  suggestedNextWorkout: z.string().max(1000).optional(),
});

const feedbackSchema = z.object({
  customerRating: z.number().int().min(1).max(5),
  customerFeedback: z.string().max(1000).optional(),
});

/** Trainer saves the workout log. Only the assigned trainer, and only once the session has started. */
export const postSessionRecord: RequestHandler = async (req, res) => {
  if (!req.auth) throw new AppError(401, 'UNAUTHENTICATED', 'Log in required');
  const body = createRecordSchema.parse(req.body);
  const bookingId = requireObjectId(body.bookingId, 'Booking');

  const booking = await Booking.findById(bookingId);
  if (!booking || booking.trainerId !== req.auth.id) {
    throw new AppError(404, 'NOT_FOUND', 'Booking not found');
  }
  if (!['IN_PROGRESS', 'COMPLETED'].includes(booking.status)) {
    throw new AppError(409, 'INVALID_STATUS', 'You can log a workout after the session has started');
  }

  const record = await SessionRecord.findOneAndUpdate(
    { bookingId },
    {
      $set: {
        customerId: booking.customerId,
        trainerId: req.auth.id,
        durationMinutes: body.durationMinutes,
        exercises: body.exercises,
        trainerNotes: body.trainerNotes,
        suggestedNextWorkout: body.suggestedNextWorkout,
      },
    },
    { upsert: true, returnDocument: 'after' },
  );

  res.status(201).json({ record });
};

/** Customer rates their own completed session. Works even if the trainer has not logged yet. */
export const postCustomerFeedback: RequestHandler = async (req, res) => {
  if (!req.auth) throw new AppError(401, 'UNAUTHENTICATED', 'Log in required');
  const bookingId = requireObjectId(req.params['bookingId'], 'Booking');
  const { customerRating, customerFeedback } = feedbackSchema.parse(req.body);

  const booking = await Booking.findById(bookingId);
  if (!booking || booking.customerId !== req.auth.id) {
    throw new AppError(404, 'NOT_FOUND', 'Booking not found');
  }
  if (booking.status !== 'COMPLETED') {
    throw new AppError(409, 'INVALID_STATUS', 'You can give feedback after the session is completed');
  }

  const record = await SessionRecord.findOneAndUpdate(
    { bookingId },
    {
      $set: { customerRating, customerFeedback },
      $setOnInsert: { customerId: booking.customerId, trainerId: booking.trainerId },
    },
    { upsert: true, returnDocument: 'after' },
  );

  res.json({ record });
};

import type { RequestHandler } from 'express';
import { AppError } from '../utils/AppError.js';
import { verifyToken } from '../utils/jwt.js';
import type { Role } from '../models/User.js';

export interface AuthUser {
  id: string;
  role: Role;
}

declare module 'express-serve-static-core' {
  interface Request {
    auth?: AuthUser;
  }
}

/** Requires a valid "Authorization: Bearer <jwt>" header. Sets req.auth. */
export const requireAuth: RequestHandler = (req, _res, next) => {
  const header = req.headers.authorization;
  if (!header || !header.startsWith('Bearer ')) {
    throw new AppError(401, 'UNAUTHENTICATED', 'Please log in to continue');
  }
  const payload = verifyToken(header.slice('Bearer '.length));
  if (!payload) {
    throw new AppError(401, 'INVALID_TOKEN', 'Your session has expired. Please log in again');
  }
  req.auth = { id: payload.sub, role: payload.role };
  next();
};

/** Use after requireAuth. Example: router.get('/x', requireAuth, requireRole('admin'), handler) */
export function requireRole(...roles: Role[]): RequestHandler {
  return (req, _res, next) => {
    if (!req.auth || !roles.includes(req.auth.role)) {
      throw new AppError(403, 'FORBIDDEN', 'You do not have access to this resource');
    }
    next();
  };
}

import type { ErrorRequestHandler, RequestHandler } from 'express';
import { ZodError } from 'zod';
import { AppError } from '../utils/AppError.js';
import { env } from '../config/env.js';

export const notFoundHandler: RequestHandler = (req, res) => {
  res.status(404).json({
    error: { code: 'NOT_FOUND', message: `Route ${req.method} ${req.originalUrl} not found` },
  });
};

function isDuplicateKeyError(err: unknown): boolean {
  return typeof err === 'object' && err !== null && 'code' in err && err.code === 11000;
}

// Express 5 forwards errors from async handlers here automatically.
export const errorHandler: ErrorRequestHandler = (err, _req, res, _next) => {
  if (err instanceof AppError) {
    res.status(err.status).json({
      error: { code: err.code, message: err.message, details: err.details },
    });
    return;
  }

  if (err instanceof ZodError) {
    res.status(400).json({
      error: {
        code: 'VALIDATION_ERROR',
        message: 'Please check the highlighted fields',
        details: err.issues.map((i) => ({ path: i.path.join('.'), message: i.message })),
      },
    });
    return;
  }

  if (isDuplicateKeyError(err)) {
    res.status(409).json({ error: { code: 'DUPLICATE', message: 'This record already exists' } });
    return;
  }

  console.error('Unhandled error:', err);
  res.status(500).json({
    error: {
      code: 'INTERNAL_ERROR',
      message: env.isProduction ? 'Something went wrong' : String(err?.message ?? err),
    },
  });
};

  id: string;
  name: string;
  email: string;
  role: Role;
  createdAt: string;
}

export function toPublicUser(user: {
  _id: { toString(): string };
  name: string;
  email: string;
  role: string;
  createdAt: Date;
}): PublicUser {
  return {
    id: user._id.toString(),
    name: user.name,
    email: user.email,
    role: user.role as Role,
    createdAt: user.createdAt.toISOString(),
  };
}

import { connectDb, disconnectDb } from '../config/db.js';
import { PricingConfig } from '../models/PricingConfig.js';
import { DEFAULT_PEAK_WINDOWS, getPricingConfig } from '../services/pricing.service.js';

async function run() {
  await connectDb();
  await getPricingConfig(); // makes sure the document exists

  await PricingConfig.updateOne(
    { _id: 'default' },
    { $set: { peakWindows: DEFAULT_PEAK_WINDOWS }, $inc: { version: 1 } },
  );

  const updated = await PricingConfig.findById('default');
  console.log(
    `Pricing windows migrated. Rates unchanged: normal ${updated?.normalPaise}, peak ${updated?.peakPaise}, commission ${updated?.commissionPercent}%. Version ${updated?.version}.`,
  );
  await disconnectDb();
}

run().catch((err) => {
  console.error('Pricing migration failed:', err);
  process.exit(1);
});

  workoutTitle: string;

// Must stay BELOW the frontend timeout for this call (55s).
const TOTAL_BUDGET_MS = 35_000;

class ProviderError extends Error {
  constructor(public status: number, message: string) {
    super(message);
  }
}

function extractAndParseJson(text: unknown): WorkoutPlan | null {
  if (typeof text !== 'string' || !text) return null;

  let cleaned = text
    .replace(/<think>[\s\S]*?<\/think>/gi, '')
    .replace(/```(?:json)?/gi, '')
    .trim();

  const first = cleaned.indexOf('{');
  const last = cleaned.lastIndexOf('}');
  if (first !== -1 && last > first) cleaned = cleaned.slice(first, last + 1);

  try {
    const parsed = JSON.parse(cleaned);
    if (
      parsed &&
      typeof parsed.workoutTitle === 'string' &&
      Array.isArray(parsed.warmUp) &&
      Array.isArray(parsed.mainCircuit) &&
      parsed.mainCircuit.length > 0
    ) {
      return {
        workoutTitle: parsed.workoutTitle,
        focusArea: parsed.focusArea || 'General Fitness',
        warmUp: parsed.warmUp.map(String),
        mainCircuit: parsed.mainCircuit.map((item: any) => ({
          exercise: String(item.exercise || 'Exercise'),
          sets: Number(item.sets) || 3,
          reps: Number(item.reps) || 10,
        })),
        coolDown: Array.isArray(parsed.coolDown) ? parsed.coolDown.map(String) : ['5 mins light stretching'],
        coachNote: parsed.coachNote || 'Maintain proper form throughout all sets.',
      };
    }
  } catch {
    // fall through
  }
  return null;
}

async function callApi(url: string, headers: Record<string, string>, body: unknown, signal: AbortSignal) {
  const response = await fetch(url, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', ...headers },
    body: JSON.stringify(body),
    signal,
  });
  if (!response.ok) {
    const errText = await response.text().catch(() => '');
    throw new ProviderError(response.status, `HTTP ${response.status}: ${errText.slice(0, 300)}`);
  }
  return (await response.json()) as any;
}

/** Tries the full body first; if the provider rejects an optional param (HTTP 400), retries with the plain body. */
async function callWithPlainRetry(
  url: string,
  headers: Record<string, string>,
  fullBody: unknown,
  plainBody: unknown,
  signal: AbortSignal,
) {
  try {
    return await callApi(url, headers, fullBody, signal);
  } catch (err) {
    if (err instanceof ProviderError && err.status === 400 && !signal.aborted) {
      return await callApi(url, headers, plainBody, signal);
    }
    throw err;
  }
}

Customer Profile:
${profileContext}

Recent Sessions:
${pastWorkoutContext}

Rules: 2-3 warm-up items, 4-5 main exercises, 2 cool-down items, coachNote under 25 words. Short strings only.
Return ONLY raw JSON (no markdown, no explanation) with exactly this schema:
{"workoutTitle":"string","focusArea":"string","warmUp":["string"],"mainCircuit":[{"exercise":"string","sets":3,"reps":10}],"coolDown":["string"],"coachNote":"string"}`;

    const nvidiaKey = env.NVIDIA_API_KEY?.trim();
    if (nvidiaKey && !nvidiaKey.includes('your-nvidia-api-key')) {

    const geminiLane: Attempt[] = [];
    const geminiKey = env.LLM_API_KEY?.trim();
    if (geminiKey && !geminiKey.includes('your-gemini-api-key') && geminiKey !== 'your_gemini_or_llm_api_key_here') {
      const candidates: { model: string; capMs: number; thinking: boolean }[] = [
        { model: env.LLM_MODEL || 'gemini-3.8-flash', capMs: 15_000, thinking: true },
        { model: 'gemini-3.5-flash-lite', capMs: 12_000, thinking: false },
        { model: 'gemini-3.6-flash', capMs: 12_000, thinking: false },
      ].filter((m, i, arr) => arr.findIndex((x) => x.model === m.model) === i);

      for (const g of candidates) {
        geminiLane.push({
          source: 'GEMINI',
          model: g.model,
          capMs: g.capMs,
          call: async (signal) => {
            const url = `https://generativelanguage.googleapis.com/v1beta/models/${g.model}:generateContent`;
            const headers = { 'x-goog-api-key': geminiKey };
            const gen = { responseMimeType: 'application/json', temperature: 0.2, maxOutputTokens: 1536 };
            const contents = [{ role: 'user', parts: [{ text: prompt }] }];
            const data = await callWithPlainRetry(
              url,
              headers,
              {
                contents,
                generationConfig: { ...gen, ...(g.thinking ? { thinkingConfig: { thinkingLevel: 'low' } } : {}) },
              },
              { contents, generationConfig: gen },
              signal,
            );
            const parts: { text?: string; thought?: boolean }[] = data.candidates?.[0]?.content?.parts ?? [];
            return parts.filter((p) => !p.thought).map((p) => p.text ?? '').join('');
          },
        });
      }
    }

    // ---- Race both lanes at once inside one deadline ------------------------------
    const controller = new AbortController();
    const deadlineTimer = setTimeout(() => controller.abort(), TOTAL_BUDGET_MS);

    async function runLane(lane: Attempt[]) {
      for (const a of lane) {
        if (controller.signal.aborted) break;
        const t0 = Date.now();
        try {
          const signal = AbortSignal.any([controller.signal, AbortSignal.timeout(a.capMs)]);
          const text = await a.call(signal);
          const plan = extractAndParseJson(text);
          if (plan) {
            console.log(`[AI] ${a.source} ${a.model} OK in ${Date.now() - t0}ms`);
            return { plan, source: a.source, model: a.model };
          }
          console.warn(`[AI] ${a.source} ${a.model} unparseable output after ${Date.now() - t0}ms`);
        } catch (err) {
          if (controller.signal.aborted) break;
          const msg = err instanceof Error ? `${err.name}: ${err.message}` : String(err);
          console.warn(`[AI] ${a.source} ${a.model} failed after ${Date.now() - t0}ms (${msg.slice(0, 200)})`);
        }
      }
      throw new Error('lane exhausted');
    }

    const lanes: Promise<{ plan: WorkoutPlan; source: 'NVIDIA' | 'GEMINI'; model: string }>[] = [];
    if (geminiLane.length) lanes.push(runLane(geminiLane));
    if (nvidiaLane.length) lanes.push(runLane(nvidiaLane));

    try {
      if (lanes.length > 0) {
        const winner = await Promise.any(lanes);
        return enrich(winner.plan, winner.source, winner.model);
      }
    } catch {
      console.warn('[AI] All AI providers failed or timed out, using rule-based plan');
    } finally {
      clearTimeout(deadlineTimer);
      controller.abort();
    }
  } catch (err) {
    console.error('[AI] Unexpected error, using rule-based plan:', err);
  }

  return fallback();
}

export interface EvaluatedCandidate {

  trainerName: string;
  distanceKm: number;
  sessionsToday: number;
  rating: number;
  eligible: boolean;
  reason?: string;

export interface SelectionResult {
  candidates: EvaluatedCandidate[];
  chosenTrainerId?: string;
  preferredHonored: boolean;
}

export function rankCandidates(
  candidates: EvaluatedCandidate[],
  preferredTrainerId?: string,
): { chosenId?: string; chosenTrainerId?: string; preferredHonored: boolean } {
  const eligible = candidates.filter((c) => c.eligible);
  if (eligible.length === 0) return { preferredHonored: false };

  if (preferredTrainerId) {
    const pref = eligible.find((c) => c.trainerId === preferredTrainerId);
    if (pref) {
      return {
        chosenId: pref.trainerId,
        chosenTrainerId: pref.trainerId,
        preferredHonored: true,
      };
    }
  }

  eligible.sort((a, b) => {
    if (a.distanceKm !== b.distanceKm) return a.distanceKm - b.distanceKm;
    if (a.sessionsToday !== b.sessionsToday) return a.sessionsToday - b.sessionsToday;
    if (b.rating !== a.rating) return b.rating - a.rating;
    return a.trainerId.localeCompare(b.trainerId);
  });

  const chosen = eligible[0]?.trainerId;
  return {
    chosenId: chosen,
    chosenTrainerId: chosen,
    preferredHonored: false,
  };
}

export async function findBestTrainerForSlot(

  const atoms = getSessionAtoms(slotStart);

  const dayKey = new Intl.DateTimeFormat('en-CA', { timeZone: 'Asia/Kolkata' }).format(slotStart);
  const startOfDay = new Date(`${dayKey}T00:00:00+05:30`);
  const endOfDay = new Date(startOfDay.getTime() + 24 * 60 * 60 * 1000 - 1);

  const evaluated: EvaluatedCandidate[] = [];

  for (const t of trainers) {
    const user = await User.findById(t.userId);
    const trainerName = user ? user.name : 'Trainer';
    const distanceKm = calculateHaversineDistance(
      society.location.lat,
      society.location.lng,
      t.baseLocation.lat,
      t.baseLocation.lng,
    );

    const sessionsToday = await Booking.countDocuments({
      trainerId: t.userId,
      startAt: { $gte: startOfDay,$lte: endOfDay },
      status: { $in: ['CONFIRMED', 'ON_THE_WAY', 'IN_PROGRESS', 'COMPLETED'] },
    });

    let eligible = true;
    let reason: string | undefined;

    const { weekday, minutes } = getIstWeekdayAndMinutes(slotStart);
    const endMinutes = minutes + 60;

    const covers = t.weeklyAvailability.some((w) => {
      if (w.weekday !== weekday) return false;
      const wStart = parseTimeToMinutes(w.start);
      const wEnd = parseTimeToMinutes(w.end);
      return minutes >= wStart && endMinutes <= wEnd;
    });

    if (!covers) {
      eligible = false;
      reason = 'Outside working hours';
    }

    if (eligible) {
      const isOff = t.timeOff.some((to) => slotStart >= to.from && slotStart < to.to);
      if (isOff) {
        eligible = false;
        reason = 'On time off';
      }
    }

    if (eligible) {
      const lockedCount = await SlotLock.countDocuments({
        trainerId: t.userId,
        atomStart: { $in: atoms },
      });
      if (lockedCount > 0) {
        eligible = false;
        reason = 'Already booked for this slot';
      }
    }

    evaluated.push({
      trainerId: t.userId,
      trainerName,
      distanceKm,
      sessionsToday,
      rating: t.rating,
      eligible,
      reason,
    });
  }

  const { chosenTrainerId, preferredHonored } = rankCandidates(evaluated, preferredTrainerId);

  return {
    candidates: evaluated,
    chosenTrainerId,
    preferredHonored,
  };
}

export interface CreateBookingInput {
  customerId: string;
  societyId: string;
  mode: BookingMode;
  startAt: Date;
  preferredTrainerId?: string;
  idempotencyKey?: string;
  paymentMode: 'PAYMENT' | 'PACKAGE';
  packageId?: string;
}

export async function createBooking(input: CreateBookingInput) {
  if (input.idempotencyKey) {
    const existing = await Booking.findOne({
      customerId: input.customerId,
      idempotencyKey: input.idempotencyKey,
    }).populate('societyId');
    if (existing) return existing;
  }

  const now = new Date();
  const minStart = new Date(now.getTime() + 10 * 60 * 1000);
  const maxHorizon = new Date(now.getTime() + 30 * 24 * 60 * 60 * 1000);

  if (input.startAt < minStart) {
    throw new AppError(400, 'INVALID_SLOT_TIME', 'Booking must be made at least 10 minutes in advance');
  }
  if (input.startAt > maxHorizon) {
    throw new AppError(400, 'INVALID_SLOT_TIME', 'Bookings cannot be made more than 30 days in advance');
  }

  if (input.mode === 'INSTANT') {
    const maxInstantTime = new Date(now.getTime() + 60 * 60 * 1000);
    if (input.startAt > maxInstantTime) {
      throw new AppError(400, 'INVALID_INSTANT_SLOT', 'Instant bookings must start within 60 minutes');
    }
  }

  const roundedStart = ceilToAtom(input.startAt);
  const roundedEnd = new Date(roundedStart.getTime() + 60 * 60 * 1000);

  const overlappingCustomerBooking = await Booking.exists({
    customerId: input.customerId,
    status: { $in: ['PENDING_PAYMENT', 'CONFIRMED', 'ON_THE_WAY', 'IN_PROGRESS'] },
    startAt: { $lt: roundedEnd },
    endAt: { $gt: roundedStart },
  });

  if (overlappingCustomerBooking) {
    throw new AppError(409, 'CUSTOMER_OVERLAP', 'You already have an active booking during this time slot');
  }

  const quote = await calculateQuoteForDate(roundedStart);
  const selection = await findBestTrainerForSlot(input.societyId, roundedStart, input.preferredTrainerId);

  if (!selection.chosenTrainerId) {
    await AssignmentDecision.create({
      requestedAt: now,
      slotStart: roundedStart,
      societyId: input.societyId,
      candidates: selection.candidates,
      outcome: 'NO_TRAINER',
    });
    throw new AppError(409, 'NO_TRAINER_AVAILABLE', 'No trainer is available for this slot. Try another time.');
  }

  const chosenTrainerId = selection.chosenTrainerId;
  const atoms = getSessionAtoms(roundedStart);

  const session = await mongoose.startSession();
  session.startTransaction();

  try {
    const bookingDoc = new Booking({
      customerId: input.customerId,
      trainerId: chosenTrainerId,
      societyId: input.societyId,
      mode: input.mode,
      startAt: roundedStart,
      endAt: roundedEnd,
      status: input.paymentMode === 'PACKAGE' ? 'CONFIRMED' : 'PENDING_PAYMENT',
      paymentMode: input.paymentMode,
      holdExpiresAt: input.paymentMode === 'PAYMENT' ? new Date(now.getTime() + 10 * 60 * 1000) : undefined,
      priceSnapshot: {
        tier: quote.tier,
        sessionPaise: quote.sessionPaise,
        commissionPaise: quote.commissionPaise,
        totalPaise: quote.totalPaise,
        commissionPercent: quote.commissionPercent,
        configVersion: quote.configVersion,
      },
      preferredTrainerId: input.preferredTrainerId,
      preferredHonored: selection.preferredHonored,
      idempotencyKey: input.idempotencyKey,
    });

    await bookingDoc.save({ session });

    const locks = atoms.map((atom) => ({
      trainerId: chosenTrainerId,
      atomStart: atom,
      bookingId: bookingDoc._id,
      state: input.paymentMode === 'PACKAGE' ? ('CONFIRMED' as const) : ('HELD' as const),
      holdExpiresAt: input.paymentMode === 'PAYMENT' ? bookingDoc.holdExpiresAt : undefined,
    }));

    await SlotLock.insertMany(locks, { session });

    if (input.paymentMode === 'PACKAGE') {
      const pkg = await consumePackageForBooking(input.customerId, bookingDoc._id.toString(), session, input.packageId);
      bookingDoc.packageId = pkg._id.toString();
      await bookingDoc.save({ session });
    }

    await AssignmentDecision.create(
      [
        {
          bookingId: bookingDoc._id,
          requestedAt: now,
          slotStart: roundedStart,
          societyId: input.societyId,
          candidates: selection.candidates,
          chosenTrainerId,
          outcome: 'ASSIGNED',
        },
      ],
      { session },
    );

    await session.commitTransaction();
    session.endSession();

    if (input.paymentMode === 'PACKAGE') {
      await sendNotification(chosenTrainerId, 'BOOKING_ASSIGNED', `New workout session assigned for ${formatIst(roundedStart)}.`, bookingDoc._id.toString());
    }

    return bookingDoc;
  } catch (err) {
    await session.abortTransaction();
    session.endSession();

    if (typeof err === 'object' && err !== null && 'code' in err && err.code === 11000) {
      throw new AppError(409, 'SLOT_ALREADY_BOOKED', 'The slot was just claimed by another booking. Please try again.');
    }
    throw err;
  }
}

const CANCELLATION_WINDOW_HOURS = 2;

export async function cancelBooking(bookingId: string, userId: string, reason?: string) {
  const session = await mongoose.startSession();
  session.startTransaction();

  try {
    const booking = await Booking.findById(bookingId).session(session);
    if (!booking) throw new AppError(404, 'NOT_FOUND', 'Booking not found');

    if (booking.customerId !== userId && booking.trainerId !== userId) {
      throw new AppError(403, 'FORBIDDEN', 'You cannot cancel this booking');
    }

    if (['CANCELLED', 'COMPLETED', 'IN_PROGRESS'].includes(booking.status)) {
      throw new AppError(400, 'INVALID_STATUS', `Cannot cancel a ${booking.status.toLowerCase()} booking`);
    }

    const now = new Date();
    const previousStatus = booking.status;
    const hoursUntilStart = (booking.startAt.getTime() - now.getTime()) / (1000 * 60 * 60);

    const eligibleForRefund = hoursUntilStart >= CANCELLATION_WINDOW_HOURS || userId === booking.trainerId;
    let restoreCredit = false;
    if (booking.paymentMode === 'PACKAGE' && booking.packageId) {
      if (eligibleForRefund) {
        restoreCredit = true;
      }
    }

    if (restoreCredit && booking.packageId) {
      await PackageLedger.create(
        [
          {
            packageId: booking.packageId,
            bookingId: booking._id,
            type: 'RESTORE',
            sessionsCount: 1,
          },
        ],
        { session },
      );

      await CustomerPackage.findByIdAndUpdate(booking.packageId, { $inc: { usedSessions: -1 } }, { session });
    }

    await SlotLock.deleteMany({ bookingId: booking._id }, { session });

    booking.status = 'CANCELLED';
    booking.cancelledAt = now;
    booking.cancelReason = reason ?? 'Cancelled by user';
    await booking.save({ session });

    let refunded = false;
    if (booking.paymentMode === 'PAYMENT' && eligibleForRefund && previousStatus !== 'PENDING_PAYMENT') {
      const payment = await Payment.findOne({ bookingId: booking._id, status: 'PAID' }).session(session);
      if (payment) {
        payment.status = 'REFUNDED';
        payment.refundedAt = now;
        payment.refundId = `demo_refund_${payment._id.toString()}`;
        await payment.save({ session });
        refunded = true;
      }
    }

    await session.commitTransaction();
    session.endSession();

    await sendNotification(
      booking.trainerId,
      'BOOKING_CANCELLED',
      `Booking for ${formatIst(booking.startAt)} was cancelled.`,
      booking._id.toString(),
    );

    await sendNotification(
      booking.customerId,
      'BOOKING_CANCELLED',
      `Your booking for ${formatIst(booking.startAt)} has been cancelled.${restoreCredit ? ' 1 package session restored.' : ''}`,
      booking._id.toString(),
    );

    return { booking, outcome: { refunded, creditRestored: restoreCredit, eligibleForRefund } };
  } catch (err) {
    await session.abortTransaction();
    session.endSession();
    throw err;
  }
}

export async function rescheduleBooking(bookingId: string, customerId: string, newStartAt: Date) {
  const booking = await Booking.findById(bookingId);
  if (!booking) throw new AppError(404, 'NOT_FOUND', 'Booking not found');

  if (booking.customerId !== customerId) {
    throw new AppError(403, 'FORBIDDEN', 'You do not own this booking');
  }

  if (booking.status !== 'CONFIRMED') {
    throw new AppError(400, 'INVALID_STATUS', 'Only confirmed bookings can be rescheduled');
  }
  if (booking.rescheduleCount >= 2) throw new AppError(409, 'RESCHEDULE_LIMIT', 'A booking can be rescheduled at most twice');
  if (booking.startAt.getTime() - Date.now() < CANCELLATION_WINDOW_HOURS * 60 * 60 * 1000) {
    throw new AppError(409, 'RESCHEDULE_WINDOW', 'Reschedule at least 2 hours before the session');
  }

  const roundedStart = ceilToAtom(newStartAt);
  const roundedEnd = new Date(roundedStart.getTime() + 60 * 60 * 1000);

  // Remove this booking's own locks before ranking so a short shift is not falsely considered busy.
  await SlotLock.deleteMany({ bookingId: booking._id });
  const selection = await findBestTrainerForSlot(
    booking.societyId.toString(),
    roundedStart,
    booking.preferredTrainerId,
  );

  if (!selection.chosenTrainerId) {
    // Restore the original locks when no candidate is available.
    await SlotLock.insertMany(getSessionAtoms(booking.startAt).map((atom) => ({ trainerId: booking.trainerId, atomStart: atom, bookingId: booking._id, state: 'CONFIRMED' as const })));
    throw new AppError(409, 'NO_TRAINER_AVAILABLE', 'No trainer available for the requested new time.');
  }

  const session = await mongoose.startSession();
  session.startTransaction();

  try {
    const newAtoms = getSessionAtoms(roundedStart);
    const locks = newAtoms.map((atom) => ({
      trainerId: selection.chosenTrainerId!,
      atomStart: atom,
      bookingId: booking._id,
      state: 'CONFIRMED' as const,
    }));

    await SlotLock.insertMany(locks, { session });

    booking.trainerId = selection.chosenTrainerId;
    booking.startAt = roundedStart;
    booking.endAt = roundedEnd;
    booking.rescheduleCount += 1;
    await booking.save({ session });

    await session.commitTransaction();
    session.endSession();

    await Promise.all([
      sendNotification(
      booking.customerId,
      'BOOKING_RESCHEDULED',
      `Your session has been rescheduled to ${formatIst(roundedStart)}.`,
      booking._id.toString(),
      ),
      sendNotification(booking.trainerId, 'BOOKING_RESCHEDULED', `Session rescheduled to ${formatIst(roundedStart)}.`, booking._id.toString()),
    ]);

    return booking;
  } catch (err) {
    await session.abortTransaction();
    session.endSession();
    throw err;
  }
}

function isDuplicateKey(err: unknown): boolean {

/**
 * Creates the package for a VERIFIED payment. Safe to call many times:
 * the unique index on paymentId guarantees one package per payment.
 */
export async function fulfilPackagePurchase(payment: HydratedDocument<IPayment>) {
  const existing = await CustomerPackage.findOne({ paymentId: payment._id });
  if (existing) return existing;

  const template = await PackageTemplate.findById(payment.packageTemplateId);
  if (!template || !template.isActive) {
    throw new AppError(404, 'NOT_FOUND', 'Package template not available');
  }

  const now = new Date();
  const expiresAt = new Date(now.getTime() + template.validityDays * 24 * 60 * 60 * 1000);

  try {
    return await CustomerPackage.create({
      customerId: payment.customerId,
      templateId: template._id,
      packageName: template.name,
      totalSessions: template.sessions,
      usedSessions: 0,
      purchasedAt: now,
      expiresAt,
      paymentId: payment._id,
    });
  } catch (err) {
    if (isDuplicateKey(err)) {
      const created = await CustomerPackage.findOne({ paymentId: payment._id });
      if (created) return created;
    }
    throw err;
  }
}

/** Atomically consumes one credit, choosing the soonest-expiring usable package by default. */
export async function consumePackageForBooking(
  customerId: string,
  bookingId: string,
  session: mongoose.ClientSession,
  requestedPackageId?: string,
) {
  const now = new Date();
  const eligibility = {
    customerId,
    expiresAt: { $gt: now },
    $expr: { $lt: ['$usedSessions', '$totalSessions'] },
  };
  const filter = requestedPackageId ? { ...eligibility, _id: requestedPackageId } : eligibility;
  const pkg = await CustomerPackage.findOneAndUpdate(filter, { $inc: { usedSessions: 1 } }, {
    session,
    sort: { expiresAt: 1 },
    returnDocument: 'after',
  });
  if (!pkg) throw new AppError(409, 'PACKAGE_UNAVAILABLE', 'No usable package credit is available');

  try {
    await PackageLedger.create([{ packageId: pkg._id, bookingId, type: 'CONSUME', sessionsCount: 1 }], { session });
  } catch (err) {
    if (isDuplicateKey(err)) throw new AppError(409, 'ALREADY_DEDUCTED', 'This booking already used a package credit');
    throw err;
  }
  return pkg;
}

/** Razorpay signature = HMAC-SHA256(secret, "orderId|paymentId"), hex encoded. */
export function computeRazorpaySignature(orderId: string, paymentId: string, secret: string): string {
  return crypto.createHmac('sha256', secret).update(`${orderId}|${paymentId}`).digest('hex');
}

/** Constant-time comparison. Anything that is not the exact HMAC (including test strings) is rejected. */
export function isValidRazorpaySignature(
  orderId: string,
  paymentId: string,
  signature: string,
  secret: string,
): boolean {
  const expected = Buffer.from(computeRazorpaySignature(orderId, paymentId, secret), 'utf8');
  const given = Buffer.from(signature, 'utf8');
  if (expected.length !== given.length) return false;
  return crypto.timingSafeEqual(expected, given);
}

function getRazorpayCredentials(): { keyId: string; keySecret: string } {
  if (!env.RAZORPAY_KEY_ID || !env.RAZORPAY_KEY_SECRET) {
    throw new AppError(503, 'PAYMENTS_NOT_CONFIGURED', 'Payments are not configured on this server');
  }
  return { keyId: env.RAZORPAY_KEY_ID, keySecret: env.RAZORPAY_KEY_SECRET };
}

export async function createPaymentOrder(customerId: string, input: CreateOrderInput) {
  const { bookingId, packageTemplateId } = input;
  if ((bookingId ? 1 : 0) + (packageTemplateId ? 1 : 0) !== 1) {
    throw new AppError(400, 'BAD_REQUEST', 'Provide exactly one of bookingId or packageTemplateId');
  }

  const { keyId, keySecret } = getRazorpayCredentials();

  let purpose: PaymentPurpose;
  let amountPaise: number;
  let receipt: string;

  if (bookingId) {
    const booking = await Booking.findById(bookingId);
    // Same message for "missing" and "someone else's" so booking IDs cannot be probed.
    if (!booking || booking.customerId !== customerId) {
      throw new AppError(404, 'NOT_FOUND', 'Booking not found');
    }
    if (booking.paymentMode !== 'PAYMENT') {
      throw new AppError(400, 'BAD_REQUEST', 'This booking is not paid by card or UPI');
    }
    if (!['PENDING_PAYMENT', 'CONFIRMED'].includes(booking.status)) {
      throw new AppError(409, 'INVALID_STATUS', `A ${booking.status.toLowerCase()} booking cannot be paid`);
    }
    if (await Payment.exists({ bookingId: booking._id, status: 'PAID' })) {
      throw new AppError(409, 'ALREADY_PAID', 'This booking is already paid');
    }
    purpose = 'BOOKING';
    amountPaise = booking.priceSnapshot.totalPaise; // always from the server-side snapshot
    receipt = `bk_${booking._id.toString()}`;
  } else {
    const template = await PackageTemplate.findById(packageTemplateId);
    if (!template || !template.isActive) throw new AppError(404, 'NOT_FOUND', 'Package template not found');
    purpose = 'PACKAGE';
    amountPaise = template.pricePaise;
    receipt = `pk_${template._id.toString()}`;
  }

  let razorpayOrderId: string;
  try {
    const client = new Razorpay({ key_id: keyId, key_secret: keySecret });
    const order = await client.orders.create({ amount: amountPaise, currency: 'INR', receipt });
    razorpayOrderId = order.id;
  } catch (err) {
    console.error('Razorpay order creation failed:', err);
    throw new AppError(502, 'PAYMENT_PROVIDER_ERROR', 'Could not start the payment. Please try again.');
  }

  const payment = await Payment.create({
    bookingId,
    packageTemplateId,
    purpose,
    customerId,
    razorpayOrderId,
    amountPaise,
    status: 'CREATED',
    signatureVerified: false,
  });

  return {
    paymentId: payment._id.toString(),
    orderId: razorpayOrderId,
    amountPaise,
    keyId,
  };
}

async function fulfilPayment(payment: InstanceType<typeof Payment>) {
  if (payment.purpose === 'PACKAGE') {

    return { payment, package: customerPackage };
  }

  if (payment.bookingId) {
    const now = new Date();
    const booking = await Booking.findOneAndUpdate(

      { $set: { status: 'CONFIRMED' }, $unset: { holdExpiresAt: 1 } },
      { returnDocument: 'after' },
    );
    if (!booking) throw new AppError(409, 'HOLD_EXPIRED', 'This payment hold expired. Please make a new booking.');
    await SlotLock.updateMany({ bookingId: booking._id, state: 'HELD' }, { $set: { state: 'CONFIRMED' }, $unset: { holdExpiresAt: 1 } });
    await sendNotification(booking.trainerId, 'BOOKING_ASSIGNED', `New workout session assigned for ${formatIst(booking.startAt)}.`, booking._id.toString());
  }
  return { payment, package: null };
}

export async function verifyPayment(customerId: string, input: VerifyInput) {
  const { orderId, paymentId, signature } = input;

  const payment = await Payment.findOne({ razorpayOrderId: orderId });
  if (!payment || payment.customerId !== customerId) {
    throw new AppError(404, 'NOT_FOUND', 'Payment order not found');
  }

  // Retry of an already verified payment: return the same result, change nothing.
  if (payment.status === 'PAID') {
    if (payment.razorpayPaymentId !== paymentId) {
      throw new AppError(409, 'ALREADY_PAID', 'This order is already paid');
    }
    return fulfilPayment(payment);
  }
  if (payment.status !== 'CREATED') {
    throw new AppError(409, 'INVALID_STATUS', `This payment is ${payment.status.toLowerCase()}`);
  }

  const { keySecret } = getRazorpayCredentials();
  if (!isValidRazorpaySignature(orderId, paymentId, signature, keySecret)) {
    throw new AppError(400, 'INVALID_SIGNATURE', 'Payment signature verification failed');
  }

  // Only one request can move CREATED to PAID, even if two verify calls arrive together.
  const claimed = await Payment.findOneAndUpdate(
    { _id: payment._id, status: 'CREATED' },
    { $set: { status: 'PAID', razorpayPaymentId: paymentId, signatureVerified: true } },
    { returnDocument: 'after' },
  );
  if (!claimed) {
    const latest = await Payment.findById(payment._id);
    if (latest && latest.status === 'PAID' && latest.razorpayPaymentId === paymentId) {
      return fulfilPayment(latest);
    }
    throw new AppError(409, 'INVALID_STATUS', 'This payment could not be verified');
  }

  try {
    return await fulfilPayment(claimed);
  } catch (err) {
    // The order has a verified payment but its booking hold is no longer usable. Preserve the audit trail.
    if (err instanceof AppError && err.code === 'HOLD_EXPIRED') {
      await Payment.updateOne({ _id: claimed._id }, { $set: { status: 'FAILED' } });
    }
    throw err;
  }
}

import type { PeakWindow } from '../models/PricingConfig.js';

const DAY_INDEX: Record<string, number> = { Sun: 0, Mon: 1, Tue: 2, Wed: 3, Thu: 4, Fri: 5, Sat: 6 };

const istParts = new Intl.DateTimeFormat('en-US', {
  timeZone: 'Asia/Kolkata',
  weekday: 'short',
  hour: 'numeric',
  minute: 'numeric',
  hourCycle: 'h23',
});

/** Weekday (0 = Sunday) and minutes since midnight of a moment, in India time. */
export function getIstWeekdayAndMinutes(date: Date): { weekday: number; minutes: number } {
  let weekdayStr = '';
  let hour = 0;
  let minute = 0;
  for (const part of istParts.formatToParts(date)) {
    if (part.type === 'weekday') weekdayStr = part.value;
    if (part.type === 'hour') hour = parseInt(part.value, 10);
    if (part.type === 'minute') minute = parseInt(part.value, 10);
  }
  return { weekday: DAY_INDEX[weekdayStr] ?? 0, minutes: hour * 60 + minute };
}

/** A slot is peak when its START time falls inside any peak window (start inclusive, end exclusive). */
export function isPeakSlot(slotStart: Date, peakWindows: PeakWindow[]): boolean {
  if (!peakWindows || peakWindows.length === 0) return false;
  const { weekday, minutes } = getIstWeekdayAndMinutes(slotStart);
  return peakWindows.some(
    (w) => w.weekdays.includes(weekday) && minutes >= w.startMinute && minutes < w.endMinute,
  );
}

export const DEFAULT_PEAK_WINDOWS: PeakWindow[] = [
  { weekdays: [1, 2, 3, 4, 5], startMinute: 360, endMinute: 540 },
  { weekdays: [1, 2, 3, 4, 5], startMinute: 1020, endMinute: 1260 },
  { weekdays: [0, 6], startMinute: 420, endMinute: 660 },
];

export interface PricingRules {
  normalPaise: number;
  peakPaise: number;
  commissionPercent: number;
  peakWindows: PeakWindow[];
  version: number;
}

export interface Quote {
  tier: 'NORMAL' | 'PEAK';
  sessionPaise: number;
  commissionPaise: number;
  totalPaise: number;
  commissionPercent: number;
  configVersion: number;
}

/** Pure function (no database). Money is integer paise; commission uses basis points to avoid float errors. */
export function buildQuote(rules: PricingRules, slotStart: Date): Quote {
  const peak = isPeakSlot(slotStart, rules.peakWindows);
  const sessionPaise = peak ? rules.peakPaise : rules.normalPaise;
  const basisPoints = Math.round(rules.commissionPercent * 100);
  const commissionPaise = Math.round((sessionPaise * basisPoints) / 10000);
  return {
    tier: peak ? 'PEAK' : 'NORMAL',
    sessionPaise,
    commissionPaise,
    totalPaise: sessionPaise + commissionPaise,
    commissionPercent: rules.commissionPercent,
    configVersion: rules.version,
  };
}

      commissionPercent: 10,
      timezone: 'Asia/Kolkata',
      peakWindows: DEFAULT_PEAK_WINDOWS,
      version: 1,
    });
  }
  return config;
}

export async function getSessionTicker(bookingId: string, auth: AuthUser) {
  const booking = await Booking.findById(bookingId);
  if (!booking) throw new AppError(404, 'NOT_FOUND', 'Booking not found');
  assertCanViewBooking(booking, auth);

  const serverTime = new Date();
  const started = booking.session?.startedAt;
  const endsAt = booking.session?.endsAt;

  if (!started || !endsAt) {
    return {
      status: booking.status,
      serverTime,
      startedAt: null,
      endsAt: null,
      elapsedSeconds: 0,
      remainingSeconds: 0,
      is55MinWarning: false,
      isCompleted: booking.status === 'COMPLETED',
    };
  }

  const warnAt = booking.session?.warnAt ?? new Date(started.getTime() + WARN_MS);
  const remainingMs = Math.max(0, endsAt.getTime() - serverTime.getTime());
  const isTimeUp = serverTime.getTime() >= endsAt.getTime();
  let status = booking.status;

  // Lazy completion (a background job takes this over in Phase 15).
  if (isTimeUp && booking.status === 'IN_PROGRESS') {
    await Booking.updateOne(
      { _id: booking._id, status: 'IN_PROGRESS' },
      { $set: { status: 'COMPLETED', 'session.completeFiredAt': serverTime } },
    );
    status = 'COMPLETED';
  }

  return {
    status,
    serverTime,
    startedAt: started,
    endsAt,
    elapsedSeconds: Math.floor((serverTime.getTime() - started.getTime()) / 1000),
    remainingSeconds: Math.floor(remainingMs / 1000),
    is55MinWarning: serverTime.getTime() >= warnAt.getTime() && !isTimeUp,
    isCompleted: isTimeUp || status === 'COMPLETED',
  };
}

export interface TimeSlotOption {
  startAt: Date;
  endAt: Date;
  availableTrainersCount: number;
  hasPreferredTrainer: boolean;
}

export interface EligibleTrainerInfo {
  trainerId: string;
  name: string;

const IST_DAY_FORMAT = new Intl.DateTimeFormat('en-CA', {
  timeZone: 'Asia/Kolkata',
  year: 'numeric',
  month: '2-digit',
  day: '2-digit',
});

export async function getAvailableSlotsForDay(societyId: string, dateStr: string, preferredTrainerId?: string) {
  const society = await Society.findById(societyId);
  if (!society) return [];

  const dayKey = /^\d{4}-\d{2}-\d{2}$/.test(dateStr)
    ? dateStr
    : IST_DAY_FORMAT.format(new Date(dateStr));
  const dayStartMs = new Date(`${dayKey}T00:00:00+05:30`).getTime();

  const now = new Date();
  const minStart = new Date(now.getTime() + 10 * 60 * 1000); // Minimum 10 mins buffer from current time

  const trainers = await TrainerProfile.find({ zoneIds: society.zoneId, isActive: true });
  const slots: TimeSlotOption[] = [];

  for (let hour = 6; hour <= 21; hour++) {
    for (let min = 0; min < 60; min += 15) {
      if (hour === 21 && min > 0) break;

      const slotStart = new Date(dayStartMs + (hour * 60 + min) * 60_000);

      // Do not return past time slots
      if (slotStart < minStart) continue;

      const slotEnd = new Date(slotStart.getTime() + 60 * 60 * 1000);

      const atoms = getSessionAtoms(slotStart);
      let availableCount = 0;
      let preferredAvailable = false;

      for (const t of trainers) {
        const { weekday, minutes } = getIstWeekdayAndMinutes(slotStart);
        const endMinutes = minutes + 60;

        const covers = t.weeklyAvailability.some((w) => {
          if (w.weekday !== weekday) return false;
          const wStart = parseTimeToMinutes(w.start);
          const wEnd = parseTimeToMinutes(w.end);
          return minutes >= wStart && endMinutes <= wEnd;
        });
        if (!covers) continue;

        const isOff = t.timeOff.some((to) => slotStart >= to.from && slotStart < to.to);
        if (isOff) continue;

        const locked = await SlotLock.exists({ trainerId: t.userId, atomStart: { $in: atoms } });
        if (locked) continue;

        availableCount++;
        if (preferredTrainerId && t.userId === preferredTrainerId) {
          preferredAvailable = true;
        }
      }

      if (availableCount > 0) {
        slots.push({
          startAt: slotStart,
          endAt: slotEnd,
          availableTrainersCount: availableCount,
          hasPreferredTrainer: preferredAvailable,
        });
      }
    }
  }

  return slots;
}

export async function getEligibleTrainersForSlot(societyId: string, startAt: Date): Promise<EligibleTrainerInfo[]> {
  const society = await Society.findById(societyId);
  if (!society) return [];

  const trainers = await TrainerProfile.find({ zoneIds: society.zoneId, isActive: true });
  const atoms = getSessionAtoms(startAt);
  const eligible: EligibleTrainerInfo[] = [];

  for (const t of trainers) {
    const { weekday, minutes } = getIstWeekdayAndMinutes(startAt);
    const endMinutes = minutes + 60;

    const covers = t.weeklyAvailability.some((w) => {
      if (w.weekday !== weekday) return false;
      const wStart = parseTimeToMinutes(w.start);
      const wEnd = parseTimeToMinutes(w.end);
      return minutes >= wStart && endMinutes <= wEnd;
    });
    if (!covers) continue;

    const isOff = t.timeOff.some((to) => startAt >= to.from && startAt < to.to);
    if (isOff) continue;

    const locked = await SlotLock.exists({ trainerId: t.userId, atomStart: { $in: atoms } });
    if (locked) continue;

    const user = await User.findById(t.userId);
    const distanceKm = calculateHaversineDistance(
      society.location.lat,
      society.location.lng,
      t.baseLocation.lat,
      t.baseLocation.lng,
    );

    eligible.push({
      trainerId: t.userId,
      name: user?.name ?? 'Trainer',
      rating: t.rating,
      bio: t.bio,
      distanceKm,
    });
  }

  return eligible;
}

export async function getInstantOptions(societyId: string) {
  const now = new Date();
  const minStart = new Date(now.getTime() + 10 * 60 * 1000);
  const roundedStart = ceilToAtom(minStart);
  const maxStart = new Date(now.getTime() + 60 * 60 * 1000);

  const options: TimeSlotOption[] = [];
  let curr = new Date(roundedStart);

  while (curr <= maxStart) {
    const trainers = await getEligibleTrainersForSlot(societyId, curr);
    if (trainers.length > 0) {
      options.push({
        startAt: new Date(curr),
        endAt: new Date(curr.getTime() + 60 * 60 * 1000),
        availableTrainersCount: trainers.length,
        hasPreferredTrainer: false,
      });
    }
    curr = new Date(curr.getTime() + 15 * 60 * 1000);
  }

  return options;
}

/** Reconciles expiring checkout holds and session lifecycle events. Safe to run repeatedly. */
export async function runTicker(now = new Date()) {
  const expired = await Booking.find({ status: 'PENDING_PAYMENT', holdExpiresAt: { $lte: now } }).select('_id');
  if (expired.length) {
    const ids = expired.map((b) => b._id);
    await Booking.updateMany({ _id: { $in: ids }, status: 'PENDING_PAYMENT' }, { $set: { status: 'EXPIRED' }, $unset: { holdExpiresAt: 1 } });
    await SlotLock.deleteMany({ bookingId: { $in: ids }, state: 'HELD' });
  }

  const completed = await Booking.find({ status: 'IN_PROGRESS', 'session.endsAt': { $lte: now }, 'session.completeFiredAt': { $exists: false } });
  for (const booking of completed) {
    const updated = await Booking.findOneAndUpdate(
      { _id: booking._id, status: 'IN_PROGRESS', 'session.completeFiredAt': { $exists: false } },
      { $set: { status: 'COMPLETED', 'session.completeFiredAt': now } },
      { returnDocument: 'after' },
    );
    if (updated) await sendNotification(updated.customerId, 'SESSION_REMINDER', 'Workout session completed! Review your trainer notes and session log.', updated._id.toString());
  }

  const warnings = await Booking.find({ status: 'IN_PROGRESS', 'session.warnAt': { $lte: now }, 'session.warnFiredAt': { $exists: false }, 'session.endsAt': { $gt: now } });
  for (const booking of warnings) {
    const updated = await Booking.findOneAndUpdate(
      { _id: booking._id, status: 'IN_PROGRESS', 'session.warnFiredAt': { $exists: false } },
      { $set: { 'session.warnFiredAt': now } }, { returnDocument: 'after' },
    );
    if (updated) {
      await Promise.all([
        sendNotification(updated.trainerId, 'SESSION_REMINDER', 'Five minutes remain in this workout session.', updated._id.toString()),
        sendNotification(updated.customerId, 'SESSION_REMINDER', 'Your workout has five minutes remaining.', updated._id.toString()),
      ]);
    }
  }
  return { expired: expired.length, completed: completed.length, warnings: warnings.length };
}

/** Helper utilities for 15-minute time atoms and slot parsing */

export const ATOM_MINUTES = 15;
export const SESSION_MINUTES = 60;
export const ATOMS_PER_SESSION = SESSION_MINUTES / ATOM_MINUTES; // 4 atoms

/** Rounds a date up to the next 15-minute interval */
export function ceilToAtom(date: Date): Date {
  const ms = 1000 * 60 * ATOM_MINUTES;
  return new Date(Math.ceil(date.getTime() / ms) * ms);
}

/** Generates array of 15-min start timestamps for a 60-min session */
export function getSessionAtoms(startAt: Date): Date[] {
  const atoms: Date[] = [];
  const startTime = startAt.getTime();
  for (let i = 0; i < ATOMS_PER_SESSION; i++) {
    atoms.push(new Date(startTime + i * ATOM_MINUTES * 60 * 1000));
  }
  return atoms;
}

/** Parses "HH:mm" time string to minutes from midnight */
export function parseTimeToMinutes(timeStr: string): number {
  const [hours, minutes] = timeStr.split(':').map((v) => parseInt(v, 10));
  return (hours ?? 0) * 60 + (minutes ?? 0);
}

/** Calculates distance in kilometers between two lat/lng points using the Haversine formula */
export function calculateHaversineDistance(lat1: number, lon1: number, lat2: number, lon2: number): number {
  const R = 6371;
  const dLat = ((lat2 - lat1) * Math.PI) / 180;
  const dLon = ((lon2 - lon1) * Math.PI) / 180;
  const a =
    Math.sin(dLat / 2) * Math.sin(dLat / 2) +
    Math.cos((lat1 * Math.PI) / 180) *
      Math.cos((lat2 * Math.PI) / 180) *
      Math.sin(dLon / 2) *
      Math.sin(dLon / 2);
  const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
  const distance = R * c;
  return Math.round(distance * 10) / 10;
}

import jwt from 'jsonwebtoken';
import { env } from '../config/env.js';
import type { Role } from '../models/User.ts';

export interface TokenPayload {
  sub: string; // user id
  role: Role;
}

export function signToken(payload: TokenPayload): string {
  return jwt.sign(payload, env.JWT_SECRET, { expiresIn: env.JWT_EXPIRES_IN_SECONDS });
}

/** Returns the payload, or null if the token is invalid or expired. */
export function verifyToken(token: string): TokenPayload | null {
  try {
    const decoded = jwt.verify(token, env.JWT_SECRET);
    if (typeof decoded === 'string') return null;
    const { sub, role } = decoded;
    if (typeof sub !== 'string' || typeof role !== 'string') return null;
    return { sub, role: role as Role };
  } catch {
    return null;
  }
}

import type { AuthUser } from '../middleware/auth.js';
import { AppError } from './AppError.js';

export interface OwnedBooking {
  customerId: string;
  trainerId: string;
}

/** Customer: own bookings. Trainer: assigned bookings. Admin: all bookings. */
export function canViewBooking(booking: OwnedBooking, auth: AuthUser): boolean {
  if (auth.role === 'admin') return true;
  if (auth.role === 'customer') return booking.customerId === auth.id;
  if (auth.role === 'trainer') return booking.trainerId === auth.id;
  return false;
}

/** Throws 404 (not 403) so we do not reveal that another user's booking exists. */
export function assertCanViewBooking(booking: OwnedBooking, auth: AuthUser): void {
  if (!canViewBooking(booking, auth)) {
    throw new AppError(404, 'NOT_FOUND', 'Booking not found');
  }
}

import express from 'express';
import cors from 'cors';
import helmet from 'helmet';
import { env } from './config/env.js';
import routes from './routes/index.js';
import { errorHandler, notFoundHandler } from './middleware/error.js';

const LOCAL_ORIGIN = /^https?:\/\/(localhost|127\.0\.0\.1|\[::1\])(:\d+)?$/;

export function createApp() {
  const app = express();

  // Render sits behind a proxy; needed so rate limiting sees the real client IP.
  app.set('trust proxy', 1);

  app.use(helmet());
  app.use(
    cors({
      // In development accept any localhost / 127.0.0.1 port (Vite may pick 5174, 5175...).
      origin: (origin, cb) => {
        if (!origin) return cb(null, true);
        if (env.clientOrigins.includes(origin)) return cb(null, true);
        if (!env.isProduction && LOCAL_ORIGIN.test(origin)) return cb(null, true);
        return cb(null, false);
      },
      methods: ['GET', 'POST', 'PUT', 'PATCH', 'DELETE', 'OPTIONS'],
      allowedHeaders: ['Content-Type', 'Authorization', 'Idempotency-Key'],
    }),
  );
  app.use(express.json({ limit: '100kb' }));

  app.use('/api', routes);

  app.use(notFoundHandler);
  app.use(errorHandler);

  return app;
}
